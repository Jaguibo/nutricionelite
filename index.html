self.respond(status, json.dumps(payload).encode(), "application/json; charset=utf-8") def do_GET(self): if self.headers.get("Host") != f"{HOST}:{PORT}": self.send_error(403) elif self.path in ("/", self.send_error(404) def do_POST(self): if self.headers.get("Host") != f"{HOST}:{PORT}" or self.headers.get("Origin") != ORIGIN: self.json_response(403, {"error": "Abrí la demo desde " + ORIGIN}) return prefix = TELEGRAM_MESSAGES.get(self.path) if prefix is None: self.send_error(404) return if self.headers.get_content_type() != "application/json": self.json_response(415, {"error": "La solicitud debe estar en formato JSON."}) return try: length = int(self.headers.get("Content-Length", "0")) if not 0 < length <= 8192: raise ValueError data = json.loads(self.rfile.read(length)) message = data.get("message") if isinstance(data, dict) else None if not isinstance(message, str) or not message.startswith(prefix + "\n") or len(message) > 3000: raise ValueError except ValueError: self.json_response(400, {"error": "El mensaje de prueba no tiene el formato esperado."}) return if simulated: self.json_response(200, {"ok": True, "simulated": True, "message_id": next(next_message_id)}) return try: result = telegram_request(token, "sendMessage", {"chat_id": chat_id, "text": message}) except TelegramError as error: self.json_response(502, {"error": str(error)}) else: self.json_response(200, {"ok": True, "message_id": result["message_id"]}) return DemoHandler def main(): parser = argparse.ArgumentParser(description=__doc__) modes = opción --chats.") chat_id = int(chat) html = DEMO_PATH.read_bytes() with ThreadingHTTPServer((HOST, PORT), handler_for(html, token, chat_id)) as server: if args.simular: print("Telegram en modo
